# === BEGIN Anti-404 WordPress (custom) ===
# Diagnostic du 19/08/2026 : 7 000 a 12 800 requetes par jour tombaient dans la
# regle WordPress finale (RewriteRule . /index.php) parce que le fichier demande
# n'existe pas. Chacune declenchait un bootstrap WordPress complet renvoyant une
# page 404 de 21 Ko, soit ~1 s de PHP et 21 Ko de bande passante pour rien.
# Principaux emetteurs : /vite.svg (favicon des landings, 3 808 fois le 18/08),
# /meta.json (crawler Facebook) et les scanners de webshells (xxx.php, wp.php...).
# Ces regles renvoient une 404 Apache immediate a la place.
#
# GARDE-FOUS (ne pas retirer) :
#  - /api/ est exclu : /web/api/.htaccess route volontairement les URL
#    inexistantes vers son proxy index.php (vers Vercel). Les court-circuiter
#    casserait l'API des landings.
#  - .txt et .xml sont exclus : robots.txt et les sitemaps sont generes
#    virtuellement par WordPress, aucun fichier physique n'existe.
#  - .html est exclu : les landings React sont servies via /landings-app/index.html.
<IfModule mod_rewrite.c>
RewriteEngine On

# 1) Asset statique inexistant -> 404 Apache directe (~0 ms au lieu de ~1 s).
RewriteCond %{REQUEST_URI} !^/api(/|$)
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule \.(svg|json|ico|png|jpe?g|gif|webp|avif|css|js|mjs|map|woff2?|ttf|otf|eot|mp4|webm|mp3)$ - [R=404,L]

# 2) Scanners de webshells : .php inexistant -> 404 immediate.
RewriteCond %{REQUEST_URI} !^/api(/|$)
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule \.php[0-9]?$ - [R=404,L]

# 3) Scanners de secrets (185.177.72.70 cherchait .env, .git/config, backup.sql).
RewriteRule (^|/)\.(env|git|svn|aws|ssh|hg)(/|$) - [F,L]
RewriteRule (^|/)[^/]*\.(sql|bak|old|swp|save|~)$ - [F,L]
RewriteRule (^|/)wp-config\.php\.[^/]+$ - [F,L]

# 4) Fichiers d'information exposes. /phpinfolws.php renvoyait 100 Ko de
#    phpinfo() en public le 19/08/2026 : version PHP, modules, chemins
#    absolus et variables d'environnement, soit une cartographie complete
#    offerte aux scanners. Le fichier reste sur le disque, seul l'acces
#    HTTP public est coupe (reversible : supprimer cette ligne).
RewriteRule (^|/)(phpinfolws\.php|phpinfo\.php|info\.php|wp-config-sample\.php)$ - [F,L]
</IfModule>
# === END Anti-404 WordPress ===

# === BEGIN Anti brute-force wp-login (custom) ===
# 1 906 POST sur //wp-login.php le 18/08 (double slash = signature du bot ;
# un navigateur legitime demande toujours /wp-login.php avec un seul slash).
# Cette regle ne peut donc PAS bloquer une connexion admin normale.
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteCond %{THE_REQUEST} \s/{2,}wp-login\.php [NC]
RewriteRule ^ - [F,L]

# IP de brute-force averees au 19/08/2026 : elles ne demandent QUE
# wp-login.php / wp-admin, aucun trafic legitime. Liste extensible.
RewriteCond %{REMOTE_ADDR} ^(217\.156\.123\.33|104\.238\.222\.26|185\.177\.72\.70)$
RewriteRule ^ - [F,L]
</IfModule>
# === END Anti brute-force wp-login ===
